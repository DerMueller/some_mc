#include <raylib.h>

int main(void)
{
    InitWindow(800, 450, "raylib example - basic window");
    SetTargetFPS(60);
    while (!WindowShouldClose())
    {
        BeginDrawing();
        ClearBackground(VIOLET);
        EndDrawing();
    }
    CloseWindow();
    return 0;
}
